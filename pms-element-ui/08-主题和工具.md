## 三、权限控制

### 权限指令

**注册指令**：

```javascript
// main.js
Vue.directive('permission', {
  inserted: function (el, binding, vnode) {
    const { value } = binding
    const permissions = store.getters && store.getters.permissions
    
    if (value && !permissions.includes(value)) {
      el.parentNode && el.parentNode.removeChild(el)
    }
  }
})
```

**使用指令**：

```vue
<el-button v-permission="'user:add'">新增</el-button>
<el-button v-permission="'user:edit'">编辑</el-button>
<el-button v-permission="'user:delete'">删除</el-button>
```

### 权限函数

```javascript
// utils/permission.js
export function checkPermission(permission) {
  const permissions = store.getters && store.getters.permissions
  return permissions.includes(permission)
}

export function checkAnyPermission(permissions) {
  return permissions.some(permission => checkPermission(permission))
}

export function checkAllPermissions(permissions) {
  return permissions.every(permission => checkPermission(permission))
}
```

**在组件中使用**：

```vue
<template>
  <div>
    <el-button v-if="hasAddPermission" @click="handleAdd">新增</el-button>
    <el-button v-if="hasEditPermission" @click="handleEdit">编辑</el-button>
  </div>
</template>

<script>
import { checkPermission } from '@/utils/permission'

export default {
  computed: {
    hasAddPermission() {
      return checkPermission('user:add')
    },
    hasEditPermission() {
      return checkPermission('user:edit')
    }
  }
}
</script>
```

### 路由权限

```javascript
// router/index.js
router.beforeEach(async (to, from, next) => {
  // 获取用户权限
  const hasPermission = checkRoutePermission(to)
  
  if (hasPermission) {
    next()
  } else {
    next({ path: '/403' })
  }
})
```

## 四、工具函数

### 格式化工具

**日期格式化**：

```javascript
// 使用全局 SDK
window.$utils.formatDate(new Date(), 'YYYY-MM-DD HH:mm:ss')

// 自定义格式化函数
export function formatDate(date, format = 'YYYY-MM-DD HH:mm:ss') {
  const d = new Date(date)
  const year = d.getFullYear()
  const month = String(d.getMonth() + 1).padStart(2, '0')
  const day = String(d.getDate()).padStart(2, '0')
  const hours = String(d.getHours()).padStart(2, '0')
  const minutes = String(d.getMinutes()).padStart(2, '0')
  const seconds = String(d.getSeconds()).padStart(2, '0')
  
  return format
    .replace('YYYY', year)
    .replace('MM', month)
    .replace('DD', day)
    .replace('HH', hours)
    .replace('mm', minutes)
    .replace('ss', seconds)
}
```

**数字格式化**：

```javascript
// 千分位格式化
export function formatNumber(num) {
  return num.toString().replace(/\B(?=(\d{3})+(?!\d))/g, ',')
}

// 金额格式化
export function formatMoney(amount, symbol = '¥', precision = 2) {
  return symbol + parseFloat(amount).toFixed(precision).replace(/\B(?=(\d{3})+(?!\d))/g, ',')
}
```

### 数据处理

**深拷贝**：

```javascript
// 使用全局 SDK
const copy = window.$utils.deepClone(original)

// 自定义深拷贝
export function deepClone(obj) {
  if (obj === null || typeof obj !== 'object') return obj
  if (obj instanceof Date) return new Date(obj)
  if (obj instanceof RegExp) return new RegExp(obj)
  
  const clone = Array.isArray(obj) ? [] : {}
  for (let key in obj) {
    if (obj.hasOwnProperty(key)) {
      clone[key] = deepClone(obj[key])
    }
  }
  return clone
}
```

**判断空值**：

```javascript
// 使用全局 SDK
window.$utils.isEmpty(value)

// 自定义判断
export function isEmpty(value) {
  return value === null || value === undefined || value === ''
}
```

### 函数工具

**防抖**：

```javascript
// 使用全局 SDK
const debouncedFn = window.$utils.debounce(fn, 500)

// 自定义防抖
export function debounce(fn, delay = 300) {
  let timer = null
  return function (...args) {
    clearTimeout(timer)
    timer = setTimeout(() => {
      fn.apply(this, args)
    }, delay)
  }
}
```

**节流**：

```javascript
// 使用全局 SDK
const throttledFn = window.$utils.throttle(fn, 500)

// 自定义节流
export function throttle(fn, delay = 300) {
  let lastTime = 0
  return function (...args) {
    const now = Date.now()
    if (now - lastTime >= delay) {
      lastTime = now
      fn.apply(this, args)
    }
  }
}
```

### URL 工具

**解析查询参数**：

```javascript
// 使用全局 SDK
const params = window.$utils.parseQuery(url)

// 自定义解析
export function parseQuery(url = window.location.href) {
  const query = {}
  const search = url.split('?')[1]
  if (search) {
    search.split('&').forEach(param => {
      const [key, value] = param.split('=')
      query[decodeURIComponent(key)] = decodeURIComponent(value || '')
    })
  }
  return query
}
```

**构建查询字符串**：

```javascript
// 使用全局 SDK
const queryString = window.$utils.buildQuery(params)

// 自定义构建
export function buildQuery(params) {
  return Object.keys(params)
    .filter(key => params[key] !== null && params[key] !== undefined)
    .map(key => `${encodeURIComponent(key)}=${encodeURIComponent(params[key])}`)
    .join('&')
}
```

## 八、ThemeProvider 运行时主题切换

**功能说明**：通过 CSS 自定义属性（变量）支持运行时主题切换，无需重新编译。

### 基础用法

```javascript
import { getThemeProvider } from 'element-ui/src/themes/theme-provider'
const themeProvider = getThemeProvider()
themeProvider.setTheme('dark')   // 切换到暗色主题
themeProvider.setTheme('light')  // 切换到亮色主题
```

### CSS 变量方式

```javascript
document.documentElement.style.setProperty('--el-color-primary', newColor)
```

### ThemeProvider API

| 方法 | 说明 |
|------|------|
| `setTheme(name, save?)` | 切换主题，`save` 默认 true（持久化到 localStorage） |
| `getTheme()` | 获取当前主题名 |
| `setVariable(name, value)` | 设置单个 CSS 变量 |
| `getVariable(name)` | 获取 CSS 变量值 |
| `setVariables(vars)` | 批量设置 CSS 变量 |
| `reset()` | 重置为默认主题 |
| `getAvailableThemes()` | 列出可用主题（如 `['light', 'dark']`） |

### 核心 CSS 变量

| 变量 | 亮色默认值 |
|------|-----------|
| `--el-color-primary` | #3C6CFE |
| `--el-color-success` | #53C31B |
| `--el-color-warning` | #F4AB1C |
| `--el-color-danger` | #FF6160 |
| `--el-color-info` | #909399 |
| `--el-color-text-primary` | #333333 |
| `--el-border-color-base` | #D9D9D9 |
| `--el-background-color-base` | #F5F7FA |

### 注意事项

- 需引入 `index-css-vars.css` 替代默认的 `index.css`
- CSS 变量采用引用链，修改 `--el-color-primary` 会自动更新所有派生变量
- 主题默认持久化到 localStorage
- **不兼容 IE**（IE 不支持 CSS 自定义属性）
